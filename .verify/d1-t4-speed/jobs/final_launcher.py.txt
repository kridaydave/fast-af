import sys,subprocess,json,time
from pathlib import Path
root=Path('/content/d1-speed/final-checks');root.mkdir(exist_ok=True);(root/'src').symlink_to(root.parent/'src',target_is_directory=True);phases=[]
for phase,cap in [('parity',180),('resume',300),('probe',630)]:
 start=time.monotonic();log=(root/f'{phase}.log').open('w')
 try:
  r=subprocess.run([sys.executable,'-u',str(root/'src/scripts/check_gdn_t4.py'),'--phase',phase,'--root',str(root),'--out',str(root/f'{phase}.json')],stdout=log,stderr=subprocess.STDOUT,timeout=cap)
  code=r.returncode
 except subprocess.TimeoutExpired:code=124
 phases.append({'phase':phase,'exit_code':code,'wall_seconds':time.monotonic()-start})
 (root/'phases.json').write_text(json.dumps(phases,indent=2))
 if code:break
if code==0:
 start=time.monotonic()
 with (root/'integration.log').open('w') as log:
  try: r=subprocess.run([sys.executable,'-u',str(root/'src/final_integration_job.py')],stdout=log,stderr=subprocess.STDOUT,timeout=180);code=r.returncode
  except subprocess.TimeoutExpired:code=124
 phases.append({'phase':'integration','exit_code':code,'wall_seconds':time.monotonic()-start})
 (root/'phases.json').write_text(json.dumps(phases,indent=2))
(root/'validation-done.json').write_text(json.dumps({'exit_code':code}))
